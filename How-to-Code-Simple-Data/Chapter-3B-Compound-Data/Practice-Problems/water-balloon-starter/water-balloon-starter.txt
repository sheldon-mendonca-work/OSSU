; When the program starts the water balloon should appear on the left side 
; of the screen, half-way up.  Since the balloon was thrown, it should 
; fly across the screen, rotating in a clockwise fashion. Pressing the 
; space key should cause the program to start over with the water balloon
; back at the left side of the screen.

; 
; NOTE: The rotate function wants an angle in degrees as its first 
; argument. By that it means Number[0, 360). As time goes by your balloon 
; may end up spinning more than once, for example, you may get to a point 
; where it has spun 362 degrees, which rotate won't accept. 
; 
; The solution to that is to use the modulo function as follows:
; 
; (rotate (modulo ... 360) (text "hello" 30 "black"))
; 
; where ... should be replaced by the number of degrees to rotate.
; 
; NOTE: It is possible to design this program with simple atomic data, 
; but we would like you to use compound data.


(define WATER-BALLOON.)
(define WIDTH 600)
(define HEIGHT 600)
(define CTR-X (/ HEIGHT 2))
(define CTR-Y (/ WIDTH 2))
(define MTS (empty-scene WIDTH HEIGHT))

(define LINEAR-SPEED 2)  
(define ROTATION-SPEED 3)

(define-struct balloon (dist rot))
;; WaterBalloon is (make-water-balloon Number Number[0, 360))
;; interp. a water ballon with is movin across the screen. It is at a distance dist from left edge at an angle rot  

(define INIT-BALLOON (make-balloon 0 0))
(define WATER-BALLOON-1 (make-balloon 90 171))
(define WATER-BALLOON-2 (make-balloon 468 359))

#;
(define (fn-for-balloon b)
  (... (balloon-dist b)     ;Number
       (student-rot b)))   ;Number[0, 360)
;; Template rules used:
;;  - compound: 2 fields

;; =================
;; Functions:

;; WaterBalloon -> WaterBalloon
;; run the animation, starting with initial balloon state.
;; Start with (main (make-balloon 0 0))

;; 
(define (main b)
  (big-bang b                   ; WaterBalloon
            (on-tick   next-balloon)     ; WaterBalloon -> WaterBalloon
            (to-draw   render-balloon)   ; WaterBalloon -> Image
            (on-key    reset-balloon)))    ; WaterBalloon KeyEvent -> WaterBalloon

;; WaterBalloon -> WaterBalloon
;; produce the next state of water balloon by rotating and translating it


(check-expect (next-balloon WATER-BALLOON-1) (make-balloon 92 174))
(check-expect (next-balloon WATER-BALLOON-2) (make-balloon 470 2))

;(define (next-balloon b) INIT-BALLOON) ;stub
(define (next-balloon b)
  (make-balloon (+ (balloon-dist b) LINEAR-SPEED) (modulo (+ (balloon-rot b) ROTATION-SPEED) 360)))


;; WaterBalloon -> Image
;; render the baloon on the screen with the give distance and rotation angle

(check-expect (render-balloon WATER-BALLOON-1)
              (place-image (rotate 171 WATER-BALLOON)
                           90
                           CTR-Y
                           MTS))
(check-expect (render-balloon WATER-BALLOON-2)
              (place-image (rotate 359 WATER-BALLOON)
                           468
                           CTR-Y
                           MTS))

;(define (render-balloon b) MTS) ;stub

(define (render-balloon b)
  (place-image (rotate (balloon-rot b) WATER-BALLOON) (balloon-dist b) CTR-Y MTS))


;; WaterBalloon KeyEvent -> WaterBalloon
;; Resets the program so the balloon is back at the intial state when space bar is pressed
(check-expect (reset-balloon (make-balloon 1 12) " ")
              (make-balloon 0 0))
(check-expect (reset-balloon (make-balloon 1 12) "left")
              (make-balloon 1 12))

; (define (reset-bs bs key) bs)

; Template from KeyEvent
(define (reset-balloon b key)
  (cond [(key=? " " key) INIT-BALLOON]
        [else b]))
